  def iter_topic_mappings() -> tuple[TopicMapping, ...]:


  262
      """Return all explicit Kafka event-topic mappings.


  263
  


  264
      This is the public discovery surface for operator tooling. Keep direct


  265
      access to `_TOPIC_MAPPING` inside this module so callers do not depend on


  266
      private registry internals.


  267
      """


  268
      return tuple(TopicMapping(event=event, topic=topic) for event, topic in _TOPIC_MAPPING.items())

  def topic_name(event: ProcessingEvent) -> str:


  272
      """Convert ProcessingEvent enum to Kafka topic string.


  273
  


  274
      Enforces explicit topic mapping. Raises ValueError with full list of mapped


  275
      events if event not in _TOPIC_MAPPING.


  276
  


  277
      Args:


  278
          event: ProcessingEvent enum value


  279
  


  280
      Returns:


  281
          Kafka topic string (e.g., "huleedu.cj_assessment.completed.v1")


  282
  


  283
      Raises:


  284
          ValueError: If event not in _TOPIC_MAPPING with list of all mapped events


  285
  


  286
      Example:


  287
          topic = topic_name(ProcessingEvent.CJ_ASSESSMENT_COMPLETED)


  288
          # Returns: "huleedu.cj_assessment.completed.v1"


  289
  


  290
      See: libs/common_core/docs/event-registry.md


  291
      """


  292
      if event not in _TOPIC_MAPPING:


  293
          mapped_events_summary = "\n".join(


  294
              [f"- {e.name} ({e.value}) ➜ '{t}'" for e, t in _TOPIC_MAPPING.items()],


  295
          )


  296
          raise ValueError(


  297
              f"Event '{event.name} ({event.value})' does not have an explicit topic mapping. "


  298
              f"All events intended for Kafka must have deliberate topic contracts defined. "


  299
              f"Currently mapped events:\n{mapped_events_summary}",


  300
          )


  301
      return _TOPIC_MAPPING[event]
